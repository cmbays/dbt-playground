// Filter kanji by JLPT level
function filterByJLPT(kanji, level) {
  if (level === 'all') return kanji;
  return kanji.filter(k => k.jlpt === level);
}

// Filter kanji by category
function filterByCategory(kanji, category) {
  if (category === 'all') return kanji;
  return kanji.filter(k => k.categories.includes(category));
}

// Filter marked kanji
function filterByMarked(kanji, markedOnly) {
  if (!markedOnly) return kanji;
  const marked = JSON.parse(localStorage.getItem('markedKanji') || '[]');
  return kanji.filter(k => marked.includes(k.character));
}

// Update card count display
function updateCardCount() {
  let filtered = homeLifeKanji;

  const jlptLevel = getSelectedJLPTLevel();
  filtered = filterByJLPT(filtered, jlptLevel);

  const category = getSelectedCategory();
  filtered = filterByCategory(filtered, category);

  const markedOnly = document.getElementById('marked-only').checked;
  filtered = filterByMarked(filtered, markedOnly);

  document.getElementById('card-count-display').textContent =
    `Ready to study ${filtered.length} kanji cards`;

  return filtered;
}

let studyState = {

    studyState.cards = shuffleArray(studyState.cards);
  }

  // Store settings

  studyState.settings.autoAudio = document.getElementById('auto-audio').checked;

  // Hide start screen, show study screen
  document.getElementById('study-start-screen').style.display = 'none';
  document.getElementById('study-active-screen').style.display = 'block';

  // Show first card
  showCard(0);
}

// Shuffle array helper
function shuffleArray(array) {
  const shuffled = [...array];
  for (let i = shuffled.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [shuffled[i], shuffled[j]] = [shuffled[j], shuffled[i]];
  }
  return shuffled;
}
```

// Toggle mark
function toggleMark() {
  const card = studyState.cards[studyState.currentIndex];
  const kanji = card.character;

  let markedKanji = JSON.parse(localStorage.getItem('markedKanji') || '[]');

  if (markedKanji.includes(kanji)) {
    // Remove mark
    markedKanji = markedKanji.filter(k => k !== kanji);
  } else {
    // Add mark
    markedKanji.push(kanji);
  }

  localStorage.setItem('markedKanji', JSON.stringify(markedKanji));
  updateMarkButton(kanji);
}

// Update mark button appearance
function updateMarkButton(kanji) {
  const markedKanji = JSON.parse(localStorage.getItem('markedKanji') || '[]');
  const isMarked = markedKanji.includes(kanji);

  const markBtn = document.getElementById('mark-btn');
  const markIcon = markBtn.querySelector('.mark-icon');

  if (isMarked) {
    markIcon.textContent = '⭐';
    markBtn.classList.add('marked');
    markBtn.innerHTML = '<span class="mark-icon">⭐</span> Marked for Review';
  } else {
    markIcon.textContent = '☆';
    markBtn.classList.remove('marked');
    markBtn.innerHTML = '<span class="mark-icon">☆</span> Mark for Review';
  }
}

// Check if kanji is marked
function isKanjiMarked(kanji) {
  const markedKanji = JSON.parse(localStorage.getItem('markedKanji') || '[]');
  return markedKanji.includes(kanji);
}

// Enhanced Japanese speech synthesis
function speakJapanese(text) {
  // Cancel any ongoing speech
  window.speechSynthesis.cancel();

  const utterance = new SpeechSynthesisUtterance(text);
  utterance.lang = 'ja-JP';
  utterance.rate = 0.8;  // Slower for learning
  utterance.pitch = 1.0;
  utterance.volume = 1.0;

  // Try to select a Japanese voice if available
  const voices = window.speechSynthesis.getVoices();
  const japaneseVoice = voices.find(voice =>
    voice.lang === 'ja-JP' || voice.lang.startsWith('ja')
  );

  if (japaneseVoice) {
    utterance.voice = japaneseVoice;
  }

  window.speechSynthesis.speak(utterance);
}

// Ensure voices are loaded before first use
window.speechSynthesis.onvoiceschanged = function() {
  window.speechSynthesis.getVoices();
};
```
